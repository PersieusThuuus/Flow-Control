    84→function matchWorkflow(meta: MetaDTO | null, categoryId: string, amount: number) {
    85→  if (!meta) return null
    86→  const matches = meta.workflows.filter(
    87→    (w) =>
    88→      w.categoryId === categoryId &&
    89→      (w.minAmount === null || amount >= w.minAmount) &&
    90→      (w.maxAmount === null || amount <= w.maxAmount),
    91→  )
    92→  return matches.length > 0 ? matches[0] : null
    93→}

    95→function slaLabel(hours: number | null): string {
    96→  if (hours === null || hours === undefined) return 'No SLA'
    97→  return hours >= 48 ? `SLA ${(hours / 24).toFixed(1)} d` : `SLA ${hours} h`
    98→}

   100→/** Decisive (non-comment) decision for a stage in the current cycle, if any. */
   101→function decisiveFor(approvals: ApprovalDTO[], cycle: number, stepIndex: number, stageId: string): ApprovalDTO | null {
   102→  return (
   103→    approvals.find(
   104→      (a) => a.cycle === cycle && a.stepIndex === stepIndex && a.stageId === stageId && a.action !== 'COMMENT',
   105→    ) ?? null
   106→  )
   107→}

   109→function computeStepState(
   110→  step: StepSnapshotDTO,
   111→  stepIndex: number,
   112→  currentStep: number,
   113→  cycleApprovals: ApprovalDTO[],
   114→): StepState {
   115→  const decisives = cycleApprovals.filter((a) => a.stepIndex === stepIndex && a.action !== 'COMMENT')
   116→  if (decisives.some((a) => a.action === 'REJECT')) return 'rejected'
   117→  if (decisives.some((a) => a.action === 'RETURN_FOR_CORRECTION')) return 'returned'
   118→  const approvedIds = new Set(decisives.filter((a) => a.action === 'APPROVE').map((a) => a.stageId))
   119→  const required = step.stages.filter((s) => s.isRequired)
   120→  const complete =
   121→    required.length > 0 ? required.every((s) => approvedIds.has(s.stageId)) : decisives.some((a) => a.action === 'APPROVE')
   122→  if (complete) return 'done'
   123→  return stepIndex === currentStep ? 'current' : 'pending'
   124→}

   126→function stepIcon(state: StepState) {
   127→  switch (state) {
   128→    case 'done':
   129→      return <CheckCircle2 className="h-4 w-4 text-emerald-600 dark:text-emerald-400" aria-hidden />
   130→    case 'current':
   131→      return <Clock className="h-4 w-4 text-amber-500" aria-hidden />
   132→    case 'rejected':
   133→      return <XCircle className="h-4 w-4 text-red-600 dark:text-red-400" aria-hidden />
   134→    case 'returned':
   135→      return <CornerUpLeft className="h-4 w-4 text-orange-500" aria-hidden />
   136→    default:
   137→      return <Circle className="h-4 w-4 text-muted-foreground/60" aria-hidden />
   138→  }
   139→}

   144→function StageDecisionLine({ approval }: { approval: ApprovalDTO | null }) {
   145→  if (!approval) return null
   146→  const behalf = approval.delegatedFromName ? ` on behalf of ${approval.delegatedFromName}` : ''
   147→  if (approval.action === 'APPROVE') {
   148→    return (
   149→      <p className="mt-2 flex items-start gap-1.5 text-xs text-emerald-600 dark:text-emerald-400">
   150→        <CheckCircle2 className="mt-0.5 h-3.5 w-3.5 shrink-0" aria-hidden />
   151→        Approved by {approval.userName}
   152→        {behalf}
   153→      </p>
   154→    )
   155→  }
   156→  if (approval.action === 'REJECT') {
   157→    return (
   158→      <p className="mt-2 flex items-start gap-1.5 text-xs text-red-600 dark:text-red-400">
   159→        <XCircle className="mt-0.5 h-3.5 w-3.5 shrink-0" aria-hidden />
   160→        Rejected by {approval.userName}
   161→        {behalf}
   162→      </p>
   163→    )
   164→  }
   165→  if (approval.action === 'RETURN_FOR_CORRECTION') {
   166→    return (
   167→      <p className="mt-2 flex items-start gap-1.5 text-xs text-orange-600 dark:text-orange-400">
   168→        <CornerUpLeft className="mt-0.5 h-3.5 w-3.5 shrink-0" aria-hidden />
   169→        Returned by {approval.userName}
   170→        {behalf}
   171→      </p>
   172→    )
   173→  }
   174→  return null
   175→}

   202→function StepTimeline({
   203→  instance,
   204→  cycleApprovals,
   205→}: {
   206→  instance: InstanceDTO
   207→  cycleApprovals: ApprovalDTO[]
   208→}) {
   209→  return (
   210→    <ol className="space-y-0">
   211→      {instance.stagePath.map((step, idx) => {
   212→        const state = computeStepState(step, idx, instance.currentStep, cycleApprovals)
   213→        const isLast = idx === instance.stagePath.length - 1
   214→        return (
   215→          <li key={idx} className="relative flex gap-3 pb-5 last:pb-0">
   216→            {!isLast ? <span className="absolute bottom-0 left-[13px] top-8 w-px bg-border" aria-hidden /> : null}
   217→            <div className="relative z-10 mt-0.5 flex h-7 w-7 shrink-0 items-center justify-center rounded-full border bg-background">
   218→              {stepIcon(state)}
   219→            </div>
   220→            <div className="min-w-0 flex-1">
   221→              <div className="flex flex-wrap items-center gap-2">
   222→                <p className="text-sm font-medium">
   223→                  Step {idx + 1}
   224→                  {step.stages.length > 1 ? ' (parallel)' : ''}
   225→                </p>
   226→                <StatusBadge
   227→                  variant={
   228→                    state === 'done'
   229→                      ? 'success'
   230→                      : state === 'current'
   231→                        ? 'warning'
   232→                        : state === 'rejected'
   233→                          ? 'destructive'
   234→                          : state === 'returned'
   235→                            ? 'warning'
   236→                            : 'secondary'
   237→                  }
   238→                >
   239→                  {STEP_STATE_LABEL[state]}
   240→                </StatusBadge>
   241→                {state === 'current' ? (
   242→                  <span className="text-xs text-muted-foreground">since {timeAgo(instance.currentStepStartedAt)}</span>
   243→                ) : null}
   244→              </div>
   245→              <div className="mt-2 grid gap-2 sm:grid-cols-2">
   246→                {step.stages.map((stage) => (
   247→                  <StageCard
   248→                    key={stage.stageId}
   249→                    stage={stage}
   250→                    approval={decisiveFor(cycleApprovals, instance.cycle, idx, stage.stageId)}
   251→                    awaiting={state === 'current'}
   252→                  />
   253→                ))}
   254→              </div>
   255→            </div>
   256→          </li>
   257→        )
   258→      })}
   259→    </ol>

   261→}

   344→function PreviousCycles({
   345→  previousInstances,

   347→}: {
   348→  previousInstances: InstanceDTO[]
   349→  earlierApprovals: ApprovalDTO[]

   361→              ({earlierApprovals.length} decision{earlierApprovals.length === 1 ? '' : 's'})
   362→            </span>
   363→          ) : null}
   364→        </span>
   365→        <ChevronDown className={`h-4 w-4 shrink-0 text-muted-foreground transition-transform ${open ? 'rotate-180' : ''}`} aria-hidden />

   391→                    <StatusBadge variant={ACTION_META[a.action].variant}>{ACTION_META[a.action].label}</StatusBadge>

   396→                  </li>
   397→                ))}
   398→            </ul>
   399→          </div>
   400→        ))}
   401→      </CollapsibleContent>
   402→    </Collapsible>
   403→  )
   404→}

   459→function VersionsTable({ docId, versions, currentVersion }: { docId: string; versions: DocumentDetailDTO['versions']; currentVersion: number }) {
   460→  const copyHash = async (hash: string) => {
   461→    try {
   462→      await navigator.clipboard.writeText(hash)
   463→      toast.success('SHA-256 hash copied to clipboard.')
   464→    } catch {
   465→      toast.error('Could not copy the hash — clipboard unavailable.')
   466→    }
   467→  }
   468→  return (
   469→    <div className="overflow-x-auto">
   470→      <Table>
   471→        <TableHeader>
   472→          <TableRow>
   473→            <TableHead className="w-16">Version</TableHead>
   474→            <TableHead className="min-w-44">File</TableHead>
   475→            <TableHead className="text-right">Size</TableHead>
   476→            <TableHead className="min-w-40">SHA-256</TableHead>
   477→            <TableHead className="min-w-32">Uploaded by</TableHead>
   478→            <TableHead className="min-w-36">Date</TableHead>
   479→            <TableHead className="min-w-40">Note</TableHead>
   480→            <TableHead className="w-24 text-right">Download</TableHead>
   481→          </TableRow>
   482→        </TableHeader>
   483→        <TableBody>
   484→          {versions.map((v) => {
   485→            const isPdf = v.mimeType === 'application/pdf'
   486→            return (
   487→              <TableRow key={v.id}>
   488→                <TableCell className="fc-num font-medium">
   489→                  v{v.version}
   490→                  {v.version === currentVersion ? (
   491→                    <Badge variant="secondary" className="ml-1.5 text-[10px]">
   492→                      latest
   493→                    </Badge>
   494→                  ) : null}
   495→                </TableCell>
   496→                <TableCell className="max-w-52">
   497→                  <p className="truncate font-medium">{v.fileName}</p>
   498→                  <p className="text-xs text-muted-foreground">{v.mimeType}</p>
   499→                </TableCell>
   500→                <TableCell className="fc-num text-right text-muted-foreground">{formatBytes(v.sizeBytes)}</TableCell>
   501→                <TableCell>
   502→                  <div className="flex items-center gap-1.5">
   503→                    <Tooltip>
   504→                      <TooltipTrigger asChild>
   505→                        <span className="fc-num font-mono text-xs">{shortHash(v.sha256)}</span>
   506→                      </TooltipTrigger>
   507→                      <TooltipContent className="max-w-72 break-all">{v.sha256}</TooltipContent>
   508→                    </Tooltip>
   509→                    <Button
   510→                      variant="ghost"
   511→                      size="sm"
   512→                      className="h-6 w-6 p-0"
   513→                      aria-label={`Copy SHA-256 of version ${v.version}`}
   514→                      onClick={() => void copyHash(v.sha256)}
   515→                    >
   516→                      <Copy className="h-3.5 w-3.5" aria-hidden />
